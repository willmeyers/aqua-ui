Get Started

Install @aqua-ui/web-components for web components or @aqua-ui/react for React.

Web Components
npm install @aqua-ui/web-components
React
npm install @aqua-ui/react
<script type="module">import "@aqua-ui/web-components";</script>

<aqua-button default>OK</aqua-button>
import { AquaWindow, AquaButton } from "@aqua-ui/react";

export const App = () => (
  <AquaWindow label="Untitled">
    <AquaButton default>OK</AquaButton>
  </AquaWindow>
);
